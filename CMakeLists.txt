cmake_minimum_required(VERSION 3.21)
project(Pass VERSION 2.0.0 LANGUAGES CXX RC)
if(MSVC)
    include("${CMAKE_BINARY_DIR}/msvc-prefix.cmake" OPTIONAL)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Widgets Test)
set(ZX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/zxcvbn-c")
add_executable(pass_dictgen "${ZX_DIR}/dict-generate.cpp")
target_compile_options(pass_dictgen PRIVATE /utf-8)
set(ZX_WORDS)
foreach(name eng_wiki female male passwd surname tv_film)
    list(APPEND ZX_WORDS "${ZX_DIR}/words-${name}.txt")
endforeach()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h"
    COMMAND pass_dictgen -o "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h" ${ZX_WORDS}
    DEPENDS pass_dictgen ${ZX_WORDS} VERBATIM)
add_library(pass_estimator src/strength_estimator.cpp "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h")
target_include_directories(pass_estimator PRIVATE "${ZX_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(pass_estimator PUBLIC Qt6::Core)
target_compile_options(pass_estimator PRIVATE /utf-8)
# Run the unmodified upstream corpus with the same embedded dictionary.
set_source_files_properties("${ZX_DIR}/test.c" "${ZX_DIR}/zxcvbn.c" PROPERTIES LANGUAGE CXX)
add_executable(pass_estimator_upstream "${ZX_DIR}/test.c" "${ZX_DIR}/zxcvbn.c" "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h")
target_include_directories(pass_estimator_upstream PRIVATE "${ZX_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(pass_estimator_safety tests/estimator_safety_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h")
target_include_directories(pass_estimator_safety PRIVATE "${ZX_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(pass_estimator_safety PRIVATE Qt6::Core)
target_compile_options(pass_estimator_safety PRIVATE /utf-8)
option(PASS_ASAN_SAFETY "Instrument the estimator safety test with AddressSanitizer" OFF)
if(PASS_ASAN_SAFETY)
    target_compile_options(pass_estimator_safety PRIVATE /fsanitize=address /Z7)
    target_link_options(pass_estimator_safety PRIVATE /DEBUG /INCREMENTAL:NO)
endif()

set(PASS_CORE_SOURCES src/password.cpp src/secure_random.cpp src/generation.cpp src/passphrase.cpp src/wordlist.cpp resources/wordlists.qrc)
add_library(pass_core ${PASS_CORE_SOURCES})
add_library(pass_core_test ${PASS_CORE_SOURCES})
target_include_directories(pass_core_test PUBLIC src)
target_link_libraries(pass_core_test PUBLIC Qt6::Core bcrypt pass_estimator)
target_compile_definitions(pass_core_test PUBLIC PASS_TESTING)
target_compile_options(pass_core_test PRIVATE /utf-8 /W4)
target_include_directories(pass_core PUBLIC src)
target_link_libraries(pass_core PUBLIC Qt6::Core bcrypt pass_estimator)
target_compile_options(pass_core PRIVATE /utf-8 /W4)
qt_add_executable(Pass WIN32 src/main.cpp resources/resources.qrc resources/app.rc)
target_link_libraries(Pass PRIVATE pass_core Qt6::Widgets)
target_compile_options(Pass PRIVATE /utf-8 /W4)
enable_testing()
add_test(NAME estimator_upstream COMMAND pass_estimator_upstream -t "${ZX_DIR}/testcases.txt")
add_test(NAME estimator_safety COMMAND pass_estimator_safety)
set_tests_properties(estimator_upstream estimator_safety PROPERTIES TIMEOUT 120)
qt_add_executable(pass_tests tests/password_test.cpp)
target_link_libraries(pass_tests PRIVATE pass_core Qt6::Test)
target_compile_options(pass_tests PRIVATE /utf-8 /W4)
add_test(NAME password_rules COMMAND pass_tests -o pass_tests-results.txt,txt)
qt_add_executable(pass_ui_tests tests/ui_test.cpp resources/resources.qrc)
target_link_libraries(pass_ui_tests PRIVATE pass_core_test Qt6::Test Qt6::Widgets)
target_compile_options(pass_ui_tests PRIVATE /utf-8 /W4)
add_test(NAME user_interface COMMAND pass_ui_tests -o pass_ui_tests-results.txt,txt)
qt_add_executable(pass_security_tests tests/security_tests.cpp)
target_compile_definitions(pass_security_tests PRIVATE PASS_CORE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/src/password.cpp")
target_include_directories(pass_security_tests PRIVATE src)
target_link_libraries(pass_security_tests PRIVATE Qt6::Core bcrypt pass_estimator)
target_compile_options(pass_security_tests PRIVATE /utf-8 /W4)
add_test(NAME security_audit COMMAND pass_security_tests)
qt_add_executable(pass_lifecycle_tests tests/lifecycle_test.cpp src/wordlist.cpp resources/wordlists.qrc)
target_compile_definitions(pass_lifecycle_tests PRIVATE PASS_TESTING)
target_include_directories(pass_lifecycle_tests PRIVATE src)
target_link_libraries(pass_lifecycle_tests PRIVATE Qt6::Test bcrypt pass_estimator)
target_compile_options(pass_lifecycle_tests PRIVATE /utf-8 /W4)
add_test(NAME secret_lifecycle COMMAND pass_lifecycle_tests -o pass_lifecycle_tests-results.txt,txt)
qt_add_executable(pass_phrase_tests tests/passphrase_test.cpp)
target_link_libraries(pass_phrase_tests PRIVATE pass_core Qt6::Test)
target_compile_options(pass_phrase_tests PRIVATE /utf-8 /W4)
add_test(NAME passphrase_rules COMMAND pass_phrase_tests -o pass_phrase_tests-results.txt,txt)
qt_add_executable(pass_phrase_security_tests tests/passphrase_security_test.cpp src/wordlist.cpp resources/wordlists.qrc)
target_include_directories(pass_phrase_security_tests PRIVATE src)
target_link_libraries(pass_phrase_security_tests PRIVATE Qt6::Test bcrypt pass_estimator)
target_compile_options(pass_phrase_security_tests PRIVATE /utf-8 /W4)
add_test(NAME passphrase_security COMMAND pass_phrase_security_tests -o pass_phrase_security_tests-results.txt,txt)
qt_add_executable(pass_missing_wordlist_tests tests/missing_wordlist_test.cpp src/password.cpp src/passphrase.cpp src/secure_random.cpp src/generation.cpp resources/resources.qrc)
target_include_directories(pass_missing_wordlist_tests PRIVATE src)
target_link_libraries(pass_missing_wordlist_tests PRIVATE Qt6::Widgets Qt6::Test bcrypt pass_estimator)
target_compile_options(pass_missing_wordlist_tests PRIVATE /utf-8 /W4)
add_test(NAME missing_wordlist COMMAND pass_missing_wordlist_tests -o pass_missing_wordlist_tests-results.txt,txt)
set_tests_properties(password_rules user_interface security_audit secret_lifecycle passphrase_rules passphrase_security missing_wordlist PROPERTIES TIMEOUT 120)

qt_add_executable(pass_capture_tests tests/capture_test.cpp resources/resources.qrc)
target_compile_definitions(pass_capture_tests PRIVATE PASS_TESTING)
target_link_libraries(pass_capture_tests PRIVATE pass_core_test Qt6::Test Qt6::Widgets)
target_compile_options(pass_capture_tests PRIVATE /utf-8 /W4)
add_test(NAME capture_protection COMMAND pass_capture_tests -o pass_capture_tests-results.txt,txt)
set_tests_properties(capture_protection PROPERTIES TIMEOUT 30)

qt_add_executable(pass_plus_tests tests/plus_test.cpp resources/resources.qrc)
target_link_libraries(pass_plus_tests PRIVATE pass_core_test Qt6::Test Qt6::Widgets)
target_compile_options(pass_plus_tests PRIVATE /utf-8 /W4)
add_test(NAME plus_features COMMAND pass_plus_tests -o pass_plus_tests-results.txt,txt)
set_tests_properties(plus_features PROPERTIES TIMEOUT 30)
