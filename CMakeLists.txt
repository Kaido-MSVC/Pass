cmake_minimum_required(VERSION 3.21)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
project(PassSSL VERSION 2.2.3 LANGUAGES CXX RC)
# Single source of truth for version and author; see src/pass_version.h.in.
set(PASS_AUTHOR "Kaido-MSVC")
set(PASS_COPYRIGHT_YEAR "2026")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/pass_version.h.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/pass_version.h" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Pass-SSL requires MSVC x64 Release /MD")
endif()
set(OPENSSL_ROOT_DIR "D:/Program Files/OpenSSL-Win64" CACHE PATH "OpenSSL 4.0.3 root")
set(OPENSSL_USE_STATIC_LIBS FALSE)
set(OPENSSL_MSVC_STATIC_RT FALSE)
set(OPENSSL_CRYPTO_LIBRARY "${OPENSSL_ROOT_DIR}/lib/VC/x64/MD/libcrypto.lib" CACHE FILEPATH "x64 MD import library" FORCE)
set(OPENSSL_INCLUDE_DIR "${OPENSSL_ROOT_DIR}/include" CACHE PATH "OpenSSL headers" FORCE)
set(LIB_EAY_RELEASE "${OPENSSL_ROOT_DIR}/lib/VC/x64/MD/libcrypto.lib" CACHE FILEPATH "Release import library" FORCE)
set(LIB_EAY_DEBUG "${OPENSSL_ROOT_DIR}/lib/VC/x64/MD/libcrypto.lib" CACHE FILEPATH "Release-only build" FORCE)
find_package(OpenSSL 4.0.3 EXACT REQUIRED COMPONENTS Crypto)
set(PASS_CRYPTO_DLL "${OPENSSL_ROOT_DIR}/libcrypto-4-x64.dll")
if(NOT EXISTS "${PASS_CRYPTO_DLL}")
    message(FATAL_ERROR "Missing OpenSSL 4 x64 runtime")
endif()
file(COPY "${PASS_CRYPTO_DLL}" DESTINATION "${CMAKE_BINARY_DIR}")
get_filename_component(PASS_COMPILER_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(PASS_DUMPBIN dumpbin HINTS "${PASS_COMPILER_DIR}" REQUIRED)
execute_process(COMMAND "${PASS_DUMPBIN}" /headers "${OPENSSL_ROOT_DIR}/lib/VC/x64/MD/libcrypto.lib" OUTPUT_VARIABLE PASS_IMPORT_HEADERS RESULT_VARIABLE PASS_IMPORT_RESULT)
if(NOT PASS_IMPORT_RESULT EQUAL 0 OR NOT PASS_IMPORT_HEADERS MATCHES "8664" OR NOT PASS_IMPORT_HEADERS MATCHES "libcrypto-4-x64.dll")
    message(FATAL_ERROR "Expected x64 libcrypto-4-x64.dll import library")
endif()
execute_process(COMMAND "${PASS_DUMPBIN}" /dependents "${PASS_CRYPTO_DLL}" OUTPUT_VARIABLE PASS_DLL_IMPORTS RESULT_VARIABLE PASS_DLL_RESULT)
if(NOT PASS_DLL_RESULT EQUAL 0 OR NOT PASS_DLL_IMPORTS MATCHES "VCRUNTIME140.dll" OR PASS_DLL_IMPORTS MATCHES "VCRUNTIME140D.dll")
    message(FATAL_ERROR "Expected release dynamic CRT OpenSSL DLL")
endif()
message(STATUS "Verified OpenSSL x64 /MD import: ${OPENSSL_CRYPTO_LIBRARY}")

if(MSVC)
    include("${CMAKE_BINARY_DIR}/msvc-prefix.cmake" OPTIONAL)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Widgets Test)
set(ZX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/zxcvbn-c")
add_executable(pass_dictgen "${ZX_DIR}/dict-generate.cpp")
target_compile_options(pass_dictgen PRIVATE /utf-8)
set(ZX_WORDS)
foreach(name eng_wiki female male passwd surname tv_film)
    list(APPEND ZX_WORDS "${ZX_DIR}/words-${name}.txt")
endforeach()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h"
    COMMAND pass_dictgen -o "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h" ${ZX_WORDS}
    DEPENDS pass_dictgen ${ZX_WORDS} VERBATIM)
add_library(pass_estimator src/strength_estimator.cpp "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h")
target_include_directories(pass_estimator PRIVATE "${ZX_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(pass_estimator PUBLIC Qt6::Core)
target_compile_options(pass_estimator PRIVATE /utf-8)
# Run the unmodified upstream corpus with the same embedded dictionary.
set_source_files_properties("${ZX_DIR}/test.c" "${ZX_DIR}/zxcvbn.c" PROPERTIES LANGUAGE CXX)
add_executable(pass_estimator_upstream "${ZX_DIR}/test.c" "${ZX_DIR}/zxcvbn.c" "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h")
target_include_directories(pass_estimator_upstream PRIVATE "${ZX_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(pass_estimator_safety tests/estimator_safety_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/dict-src.h")
target_include_directories(pass_estimator_safety PRIVATE "${ZX_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(pass_estimator_safety PRIVATE Qt6::Core)
target_compile_options(pass_estimator_safety PRIVATE /utf-8)
option(PASS_ASAN_SAFETY "Instrument the estimator safety test with AddressSanitizer" OFF)
if(PASS_ASAN_SAFETY)
    target_compile_options(pass_estimator_safety PRIVATE /fsanitize=address /Z7)
    target_link_options(pass_estimator_safety PRIVATE /DEBUG /INCREMENTAL:NO)
endif()
